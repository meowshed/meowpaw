<role>
A research record says what was read and what it showed, so a requirement can
cite evidence instead of an argument.
</role>

<rules name="sections, in order">
<rule id="summary">A summary answering the question in five to eight
sentences.</rule>
<rule id="question">The question, and why it needed answering.</rule>
<rule id="method">The method: what was read or run, when, and the limits of
that.</rule>
<rule id="findings">The findings, each naming the source it comes from, read at
the primary source and not a summary of it.</rule>
<rule id="conclusions">Numbered conclusions, each one a requirement could
cite.</rule>
<rule id="sources">The sources, each with what it supported.</rule>
</rules>

<rules name="done">
<rule id="done">It is done when every conclusion traces to a finding and every
finding to a source.</rule>
</rules>
