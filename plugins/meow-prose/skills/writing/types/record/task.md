<role>
A task is one unit of work: one branch, one change, one review.
</role>

<rules name="sections, in order">
<rule id="what">What to do, in enough detail for an implementer with no access
to the conversation, with the constraints that bind it.</rule>
<rule id="depends">What it depends on, and why: a dependency that is only
convenience is not one.</rule>
<rule id="evidence">The evidence once done: the command, its result, and the
change that carried it.</rule>
<rule id="left-alone">What it left alone, such as documentation deliberately
not updated, and why.</rule>
</rules>

<rules name="done">
<rule id="done">It is done when somebody else could verify it from its evidence
alone.</rule>
</rules>
