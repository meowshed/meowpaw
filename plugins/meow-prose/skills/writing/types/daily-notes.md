<role>
Daily notes record a day's work for the author a week later and for whoever
picks the work up.
</role>

<rules name="sections, in order">
<rule id="order">Write what you did, what you learned and what is blocked, in
that order, one line each.</rule>
<rule id="reason">Let a line be a fragment, and still name the thing and the
reason: "Reverted the retry change, because the two events it merged meant
different things", not "reverted retry change".</rule>
</rules>

<rules name="done">
<rule id="done">It is done when each line can be acted on a week later.</rule>
</rules>
