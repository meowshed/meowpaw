<role>
A changelog is the running record of changes, version by version, read by
somebody looking for when a thing changed.
</role>

<rules name="sections, in order">
<rule id="version">Head each version with its number and its date as
YYYY-MM-DD.</rule>
<rule id="groups">Group entries under Added, Changed, Deprecated, Removed,
Fixed and Security, so a reader scanning for one kind finds it.</rule>
<rule id="entry">Write one sentence per entry, naming the change, with no
marketing language.</rule>
</rules>

<rules name="done">
<rule id="done">It is done when a reader can find the version in which any
change landed.</rule>
</rules>
