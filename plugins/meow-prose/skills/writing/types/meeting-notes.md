<role>
Meeting notes record what a meeting decided for the people who were not there.
</role>

<rules name="sections, in order">
<rule id="header">Open with the date and who was there.</rule>
<rule id="decisions">List the decisions, each with who made it and why in one
sentence.</rule>
<rule id="questions">List the open questions.</rule>
<rule id="actions">List the actions, each with an owner and a date, because an
action with no owner is not done.</rule>
<rule id="discussion">Record discussion only where it explains a decision,
because nobody rereads the back-and-forth.</rule>
</rules>

<rules name="done">
<rule id="done">It is done when somebody who missed the meeting knows what was
decided and what they owe.</rule>
</rules>
